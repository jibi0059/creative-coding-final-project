let mechanicToggles = { audio: true, perlin: true, input: true, time: true }; function setup() { createCanvas(windowWidth, windowHeight); setupMechanicToggles(); if (typeof setupAudioMechanic === "function") { setupAudioMechanic(); } if (typeof setupPerlinMechanic === "function") { setupPerlinMechanic(); } if (typeof setupInputMechanic === "function") { setupInputMechanic(); } if (typeof setupTimeMechanic === "function") { setupTimeMechanic(); } } function draw() { background(8, 12, 28); if (mechanicToggles.time && typeof drawTimeMechanic === "function") { drawTimeMechanic(); } if (mechanicToggles.perlin && typeof drawPerlinMechanic === "function") { drawPerlinMechanic(); } if (mechanicToggles.input && typeof drawInputMechanic === "function") { drawInputMechanic(); } if (mechanicToggles.audio && typeof drawAudioMechanic === "function") { drawAudioMechanic(); } } function windowResized() { resizeCanvas(windowWidth, windowHeight); if (typeof resizeAudioMechanic === "function") { resizeAudioMechanic(); } if (typeof resizePerlinMechanic === "function") { resizePerlinMechanic(); } if (typeof resizeInputMechanic === "function") { resizeInputMechanic(); } if (typeof resizeTimeMechanic === "function") { resizeTimeMechanic(); } } function setupMechanicToggles() { const audioToggle = document.getElementById("toggle-audio"); const perlinToggle = document.getElementById("toggle-perlin"); const inputToggle = document.getElementById("toggle-input"); const timeToggle = document.getElementById("toggle-time"); if (audioToggle) { mechanicToggles.audio = audioToggle.checked; audioToggle.addEventListener("change", function () { mechanicToggles.audio = audioToggle.checked; }); } if (perlinToggle) { mechanicToggles.perlin = perlinToggle.checked; perlinToggle.addEventListener("change", function () { mechanicToggles.perlin = perlinToggle.checked; }); } if (inputToggle) { mechanicToggles.input = inputToggle.checked; inputToggle.addEventListener("change", function () { mechanicToggles.input = inputToggle.checked; }); } if (timeToggle) { mechanicToggles.time = timeToggle.checked; timeToggle.addEventListener("change", function () { mechanicToggles.time = timeToggle.checked; }); } }
